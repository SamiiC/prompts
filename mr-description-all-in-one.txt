#!/usr/bin/env python3
"""Render figures in the "engineering blog" style (black card, mono caps label, bold
headline, takeaway, legend, Replay/Pause, thin white outlines, periwinkle fills).

Four figure kinds, chosen with "kind" in the spec JSON:
  flow      request flow between boxes, animated request/response particles
  timeline  Gantt-style lanes over time, animated scrubber, one or more panels
  stepper   tabbed state walkthrough (e.g. LIFO vs FIFO connection pools)
  lanes     stacked comparison panels of pooled/multiplexed connection lanes

Usage:
  render_diagram.py spec.json --out-dir OUT [--name figure-01] [--png] [--scale N]
                    [--freeze-at SECONDS] [--step N] [--keep-buttons]
Outputs OUT/<name>.svg and .html (animated/interactive) and, with --png, a static .png.
Spec format and design rules: references/diagram-style.md
"""
import argparse, html, json, math, os, sys

SANS = "'Soehne','Helvetica Neue',Inter,'Segoe UI',Arial,'DejaVu Sans',sans-serif"
MONO = "'Soehne Mono','JetBrains Mono','SF Mono',Menlo,Consolas,'DejaVu Sans Mono',monospace"
BG, BORDER, RULE, NODE, EDGE, WHITE, DOT = "#000", "#2b2b2b", "#2b2b2b", "#d6d6d6", "#f2f2f2", "#fff", "#3a3a3a"
COL = {"req": "#6f9cf7", "res": "#9be8c6", "cpu": "#8fa9ee", "net": "#5b7bd6", "wait": "#bcd0f5",
       "hl": "#8fb0f5", "ink": "#0b1020"}
KIND_DEFAULTS = {  # width, font scale, side padding
    "flow": (1940, 1.0, 64), "timeline": (1790, 0.9, 64),
    "stepper": (1620, 0.9, 64), "lanes": (1176, 0.62, 50),
}


def esc(s): return html.escape(str(s), quote=True)
def col(c): return COL.get(c, c)


def T(x, y, s, size=20, fam=SANS, w=400, fill=WHITE, anchor="start", ls=None, extra=""):
    ls = f' letter-spacing="{ls}"' if ls is not None else ""
    return (f'<text x="{x:.1f}" y="{y:.1f}" fill="{fill}" font-family="{fam}" font-size="{size:.1f}" '
            f'font-weight="{w}" text-anchor="{anchor}"{ls}{extra}>{esc(s)}</text>')


def MT(x, y, s, size=11, fill=WHITE, anchor="start", extra=""):
    return T(x, y, str(s).upper(), size, MONO, 700, fill, anchor, 0.3, extra)


def rect(x, y, w, h, rx=0, fill="none", stroke=None, sw=1, dash=None, extra=""):
    st = f' stroke="{stroke}" stroke-width="{sw}"' if stroke else ""
    da = f' stroke-dasharray="{dash}"' if dash else ""
    return (f'<rect x="{x:.1f}" y="{y:.1f}" width="{w:.1f}" height="{h:.1f}" rx="{rx}" fill="{fill}"{st}{da}{extra}/>')


def wrap(s, n):
    words, lines, cur = s.split(), [], ""
    for w in words:
        if len(cur) + len(w) + 1 > n and cur:
            lines.append(cur); cur = w
        else:
            cur = (cur + " " + w).strip()
    return lines + ([cur] if cur else [])

def legend_icon(shape, color, x, cy, s):
    c = col(color)
    if shape == "square":
        return rect(x, cy - 10.5 * s, 21 * s, 21 * s, 3, c, "#c9d6fb"), 21 * s
    if shape == "ring":
        return f'<circle cx="{x + 10.5 * s:.1f}" cy="{cy:.1f}" r="{9.5 * s:.1f}" fill="{BG}" stroke="{WHITE}" stroke-width="1.2"/>', 21 * s
    if shape == "dash":
        return f'<line x1="{x:.1f}" y1="{cy:.1f}" x2="{x + 26 * s:.1f}" y2="{cy:.1f}" stroke="{WHITE}" stroke-dasharray="3 3"/>', 26 * s
    return f'<circle cx="{x + 10.5 * s:.1f}" cy="{cy:.1f}" r="{10.5 * s:.1f}" fill="{c}" stroke="#d6e2fb"/>', 21 * s

# ----------------------------------------------------------------- flow ----
def anchor(n, side):
    x, y, w, h = n["x"], n["y"], n["w"], n["h"]
    return {"r": (x + w, y + h / 2), "l": (x, y + h / 2), "t": (x + w / 2, y), "b": (x + w / 2, y + h)}[side]


def route(a, b, sides, vx=None, vy=None, via=None):
    if via: return [a] + [tuple(p) for p in via] + [b]
    if sides[0] in "lr":
        if abs(a[1] - b[1]) < 0.6: return [a, b]
        v = vx if vx is not None else (a[0] + b[0]) / 2
        return [a, (v, a[1]), (v, b[1]), b]
    if abs(a[0] - b[0]) < 0.6: return [a, b]
    v = vy if vy is not None else (a[1] + b[1]) / 2
    return [a, (a[0], v), (b[0], v), b]


def rpath(pts, r=8):
    c = [pts[0]]
    for p in pts[1:]:
        if abs(p[0] - c[-1][0]) > .01 or abs(p[1] - c[-1][1]) > .01: c.append(p)
    pts = c
    d = f"M{pts[0][0]:.1f},{pts[0][1]:.1f}"
    for i in range(1, len(pts) - 1):
        p0, p1, p2 = pts[i - 1], pts[i], pts[i + 1]
        l1, l2 = math.hypot(p1[0] - p0[0], p1[1] - p0[1]), math.hypot(p2[0] - p1[0], p2[1] - p1[1])
        rr = min(r, l1 / 2, l2 / 2)
        ux, uy, vx, vy = (p1[0] - p0[0]) / l1, (p1[1] - p0[1]) / l1, (p2[0] - p1[0]) / l2, (p2[1] - p1[1]) / l2
        d += f" L{p1[0] - ux * rr:.1f},{p1[1] - uy * rr:.1f} Q{p1[0]:.1f},{p1[1]:.1f} {p1[0] + vx * rr:.1f},{p1[1] + vy * rr:.1f}"
    return d + f" L{pts[-1][0]:.1f},{pts[-1][1]:.1f}"


def particle(kind, href, dur, begin, ring=False, size=1.0):
    """A request square / response dot moving along a path, looping forever."""
    if kind == "req":
        sh, close = (f'<rect x="{-6.5 * size}" y="{-6.5 * size}" width="{13 * size}" height="{13 * size}" rx="2.5" '
                     f'fill="{COL["req"]}" stroke="#c9d6fb">'), "</rect>"
    elif ring:
        sh, close = f'<circle r="{6.5 * size}" fill="{BG}" stroke="{WHITE}" stroke-width="1.2">', "</circle>"
    else:
        sh, close = f'<circle r="{6.5 * size}" fill="{COL["res"]}" stroke="#c5f3df">', "</circle>"
    return (sh + f'<animateMotion dur="{dur:.2f}s" begin="{begin:.2f}s" repeatCount="indefinite" rotate="0">'
            f'<mpath xlink:href="{href}"/></animateMotion>' + close)


def body_flow(spec, W, R):
    H = spec.get("height", 920)
    nodes = {n["id"]: n for n in spec.get("nodes", [])}
    edges = spec.get("edges", [])
    steps = sorted({e.get("step", 1) for e in edges if e.get("particles", True)}) or [1]
    n = max(steps); seg = 1.1; win = seg * 2.8; Tt = 2 * n * seg + win + 1.0
    out = [f'<rect x="16" y="{R:.1f}" width="{W - 32}" height="{H - 16 - R:.1f}" fill="url(#dots)"/>']
    for g in spec.get("groups", []):
        out.append(rect(g["x"] + .5, g["y"] + .5, g["w"], g["h"], 2, BG, NODE))
    for lb in spec.get("labels", []):
        out.append(MT(lb["x"], lb["y"], lb["text"], 13, anchor="middle"))
    defs, parts = [], []
    for i, e in enumerate(edges):
        fs_, ts_ = e.get("fromSide", "r"), e.get("toSide", "l")
        a, b = anchor(nodes[e["from"]], fs_), anchor(nodes[e["to"]], ts_)
        if "fromY" in e: a = (a[0], e["fromY"])
        if "toY" in e: b = (b[0], e["toY"])
        pts = route(a, b, (fs_, ts_), e.get("viaX"), e.get("viaY"), e.get("via"))
        ar = e.get("arrows", "both")
        ms = ' marker-start="url(#chev)"' if ar in ("both", "start") else ""
        me = ' marker-end="url(#chev)"' if ar in ("both", "end") else ""
        out.append(f'<path id="e{i}" d="{rpath(pts)}" fill="none" stroke="{EDGE}" stroke-width="1"{ms}{me}/>')
        defs.append(f'<path id="e{i}r" d="{rpath(list(reversed(pts)))}"/>')
        if e.get("particles", True):
            k = e.get("step", 1)
            for kind, t0 in (("req", (k - 1) * seg), ("res", (2 * n - k) * seg)):
                a_, b_ = t0 / Tt, min((t0 + win) / Tt, 1.0)
                ep = 0.12 * win / Tt
                sh = (f'<rect x="-6.5" y="-6.5" width="13" height="13" rx="2.5" fill="{COL["req"]}" stroke="#c9d6fb" opacity="0">'
                      if kind == "req" else f'<circle r="6.5" fill="{COL["res"]}" stroke="#c5f3df" opacity="0">')
                close = "</rect>" if kind == "req" else "</circle>"
                parts.append(sh + f'<animateMotion dur="{Tt:.2f}s" repeatCount="indefinite" calcMode="linear" keyPoints="0;0;1;1" '
                             f'keyTimes="0;{a_:.4f};{b_:.4f};1" rotate="0"><mpath xlink:href="#{"e%d" % i if kind == "req" else "e%dr" % i}"/></animateMotion>'
                             f'<animate attributeName="opacity" dur="{Tt:.2f}s" repeatCount="indefinite" values="0;0;1;1;0;0" '
                             f'keyTimes="0;{a_:.4f};{min(a_ + ep, b_):.4f};{max(b_ - ep, a_):.4f};{b_:.4f};1"/>' + close)
    out.append("<defs>" + "".join(defs) + "</defs>")
    for nd in spec.get("nodes", []):
        x, y, w, h = nd["x"], nd["y"], nd["w"], nd["h"]
        out.append(rect(x + .5, y + .5, w, h, 3, BG, NODE))
        if nd.get("sub"):
            out.append(T(x + 19, y + h / 2 - 1, nd["label"], 20))
            out.append(MT(x + 19, y + h / 2 + 20, nd["sub"], 13))
        else:
            out.append(T(x + 19, y + h / 2 + 7, nd["label"], 20))
    out += parts
    return out, H, {"T": Tt, "freeze": n * seg * 0.95}

def body_timeline(spec, W, R, pad):
    ax = spec.get("axis", {"max": 40, "ticks": [0, 10, 20, 30, 40]})
    tmax = ax["max"]; X0, X1 = pad + 186, W - pad; px = (X1 - X0) / tmax
    play = tmax * 0.3; Tt = play + 1.6
    out, y, cid = [], R, 0

    def reveal(t0, t1):
        a, b = t0 / tmax * play / Tt, t1 / tmax * play / Tt
        return a, b

    def track(ty, label_y, segs):
        nonlocal cid
        cid += 1
        o = [f'<clipPath id="tc{cid}">{rect(X0 + .5, ty + .5, X1 - X0, 44, 22)}</clipPath><g clip-path="url(#tc{cid})">']
        if segs and segs[0]["t"] > 0:
            o.append(f'<line x1="{X0 + 12}" y1="{ty + 22}" x2="{X0 + segs[0]["t"] * px:.1f}" y2="{ty + 22}" stroke="#777" stroke-dasharray="1 3"/>')
        for s in segs:
            x, w = X0 + s["t"] * px, s["d"] * px
            k = s.get("kind", "cpu")
            fill = {"cpu": COL["cpu"], "net": COL["net"], "wait": COL["wait"]}.get(k, BG)
            dash = "2 2" if k in ("wait", "blocked") else None
            stroke = WHITE if k in ("wait", "blocked") else ("#9a9a9a" if k == "plain" else "#000")
            a, b = reveal(s["t"], s["t"] + s["d"])
            anim = (f'<animate attributeName="width" dur="{Tt:.2f}s" repeatCount="indefinite" calcMode="linear" '
                    f'values="0;0;{w:.1f};{w:.1f}" keyTimes="0;{a:.4f};{b:.4f};1"/>') if k in ("cpu", "net", "wait") else ""
            o.append(f'<rect x="{x:.1f}" y="{ty + 1}" width="{w:.1f}" height="42" fill="{fill}" stroke="{stroke}" '
                     f'stroke-width="{1 if dash else 1.2}"{" stroke-dasharray=%s" % chr(34) + dash + chr(34) if dash else ""}>{anim}</rect>')
            if s.get("text") and w > len(s["text"]) * 8 + 8:
                tc = COL["ink"] if k in ("cpu", "wait") else WHITE
                o.append(MT(x + w / 2, ty + 26, s["text"], 11, tc, "middle"))
        o.append("</g>" + rect(X0 + .5, ty + .5, X1 - X0, 44, 22, "none", WHITE))
        return "".join(o)

    for pi, p in enumerate(spec["panels"]):
        py = y
        if pi > 0: out.append(f'<line x1="{pad}" y1="{py:.1f}" x2="{W - pad}" y2="{py:.1f}" stroke="{RULE}"/>')
        out.append(T(pad, py + 46, p["title"], 23, SANS, 600, ls=-0.2))
        if p.get("note"): out.append(T(W - pad, py + 46, p["note"], 18, anchor="end"))
        if p.get("status"): out.append(T(pad, py + 91, p["status"], 18))
        out.append(MT(pad, py + 152, ax.get("label", "Time \u2192"), 11))
        for tk in ax["ticks"]:
            out.append(MT(X0 + tk * px, py + 152, tk, 11, anchor="end" if tk == tmax else "start"))
        thr = p["thread"]
        out.append(T(pad, py + 176 + 29, thr.get("label", "Python thread"), 20, SANS, 500))
        out.append(track(py + 176, 0, thr["segments"]))
        out.append(f'<line x1="{pad}" y1="{py + 240}" x2="{W - pad}" y2="{py + 240}" stroke="{RULE}"/>')
        ly = py + 276
        for ln in p["lanes"]:
            out.append(T(pad, ly + 16, ln["label"], 20, SANS, 500))
            for j, sl in enumerate(wrap(ln.get("sub", "").upper(), 20)[:2]):
                out.append(MT(pad, ly + 36 + j * 17, sl, 11))
            out.append(track(ly, 0, ln["segments"]))
            if "marker" in ln:
                mx = X0 + ln["marker"] * px
                out.append(f'<line x1="{mx:.1f}" y1="{ly - 2}" x2="{mx:.1f}" y2="{ly + 46}" stroke="{WHITE}" stroke-width="2"/>'
                           f'<polygon points="{mx:.1f},{ly - 10} {mx + 5:.1f},{ly - 5} {mx:.1f},{ly} {mx - 5:.1f},{ly - 5}" fill="{WHITE}"/>')
            ly += 87
        out.append(f'<line x1="{X0}" y1="{py + 166}" x2="{X0}" y2="{ly - 40}" stroke="#4a6fd0" stroke-width="1">'
                   f'<animate attributeName="x1" dur="{Tt:.2f}s" repeatCount="indefinite" values="{X0};{X1};{X1}" keyTimes="0;{play / Tt:.4f};1"/>'
                   f'<animate attributeName="x2" dur="{Tt:.2f}s" repeatCount="indefinite" values="{X0};{X1};{X1}" keyTimes="0;{play / Tt:.4f};1"/></line>')
        if p.get("footer"): out.append(T(X0, ly - 87 + 108, p["footer"], 19))
        y = ly - 87 + 108 + 62
    return out, y, {"T": Tt, "freeze": play * 0.46}

#!/usr/bin/env bash
# Collect everything needed to describe the current branch against its base.
# Usage: collect_diff.sh [base-branch] [--max-lines N] [--working-tree]
#   base-branch      defaults to origin/HEAD, then main, master, develop (first that exists)
#   --max-lines N    cap on full-diff lines printed (default 1500)
#   --working-tree   also include uncommitted + staged changes
set -euo pipefail

BASE=""
MAX_LINES=1500
WORKTREE=0
while [[ $# -gt 0 ]]; do
  case "$1" in
    --max-lines) MAX_LINES="$2"; shift 2 ;;
    --working-tree) WORKTREE=1; shift ;;
    *) BASE="$1"; shift ;;
  esac
done

git rev-parse --is-inside-work-tree >/dev/null 2>&1 || { echo "not a git repo" >&2; exit 1; }

if [[ -z "$BASE" ]]; then
  ORIGIN_HEAD=$(git symbolic-ref --quiet --short refs/remotes/origin/HEAD 2>/dev/null || true)
  for cand in "$ORIGIN_HEAD" main master develop origin/main origin/master; do
    if [[ -n "$cand" ]] && git rev-parse --verify --quiet "$cand" >/dev/null; then
      BASE="$cand"; break
    fi
  done
fi
[[ -n "$BASE" ]] || { echo "could not find a base branch; pass one explicitly" >&2; exit 1; }

HEAD_REF=$(git rev-parse --abbrev-ref HEAD)
MERGE_BASE=$(git merge-base "$BASE" HEAD)

# Noise that never helps a human reviewer understand intent.
EXCLUDES=(':(exclude)*.lock' ':(exclude)package-lock.json' ':(exclude)yarn.lock'
          ':(exclude)pnpm-lock.yaml' ':(exclude)poetry.lock' ':(exclude)Cargo.lock'
          ':(exclude)go.sum' ':(exclude)*.min.js' ':(exclude)*.snap'
          ':(exclude)dist/*' ':(exclude)build/*' ':(exclude)vendor/*')

if [[ $WORKTREE -eq 1 ]]; then RANGE_ARGS=("$MERGE_BASE"); else RANGE_ARGS=("$MERGE_BASE" HEAD); fi

echo "=== BRANCH ==="
echo "head:       $HEAD_REF"
echo "base:       $BASE"
echo "merge-base: $(git rev-parse --short "$MERGE_BASE")"
echo
echo "=== COMMITS (oldest first) ==="
git log --reverse --format='%h %s%n%w(0,4,4)%b' "$MERGE_BASE"..HEAD | sed '/^\s*$/d'
echo
echo "=== FILES (status) ==="
git diff --name-status -M "${RANGE_ARGS[@]}" -- . "${EXCLUDES[@]}"
echo
echo "=== DIFFSTAT ==="
git diff --stat=100 -M "${RANGE_ARGS[@]}" -- . "${EXCLUDES[@]}"
echo
echo "=== SHORTSTAT ==="
git diff --shortstat -M "${RANGE_ARGS[@]}" -- . "${EXCLUDES[@]}"
echo

TOTAL=$(git diff -M -U3 "${RANGE_ARGS[@]}" -- . "${EXCLUDES[@]}" | wc -l)
echo "=== FULL DIFF (${TOTAL} lines; showing up to ${MAX_LINES}) ==="
git diff -M -U3 "${RANGE_ARGS[@]}" -- . "${EXCLUDES[@]}" | head -n "$MAX_LINES"
if (( TOTAL > MAX_LINES )); then
  echo
  echo "... truncated. Read the remaining files individually with:"
  echo "    git diff -M -U3 $MERGE_BASE HEAD -- <path>"
fi

# Diagram style: the Habitat figures

Every figure is a dark card: mono caps label (`FIGURE 03 · TOPIC`), bold headline, one- or two-sentence takeaway, an optional legend, Replay and Pause buttons, a hairline rule, then the body. Fills are periwinkle blues, outlines are thin white, text is white, mono caps are used for every small label. Request = blue square, response = mint circle (or white ring in `lanes`).

Never hand-draw SVG. Write a JSON spec and run `scripts/render_diagram.py`. The renderer owns the look, so every figure in every MR is identical in style. Each kind has a working 1:1 replica of a source figure in `assets/`: copy the closest one and edit it.

## Contents
1. Pick the kind
2. Commands
3. Shared fields
4. `flow`: request flow between components
5. `timeline`: who is doing what, over time
6. `stepper`: tabbed walkthrough of how state changes
7. `lanes`: stacked variants of one flow
8. Content rules (all kinds)
9. Captions
10. Checklist

---

## 1. Pick the kind

Decide from what the MR changes, not from what looks fanciest. Use one figure by default and two at most, and the two may be different kinds.

| The MR changes... | Kind | Template | Source figure |
|:--|:--|:--|:--|
| Which components talk to which: a new service, a proxy, a router, a split, a migration, new storage | `flow` | `assets/example-spec.json` | Fig 02, "Simplified Habitat request flow" |
| Ordering, scheduling, blocking, concurrency, locks, retries, timeouts, queueing delay, async vs sync | `timeline` | `assets/example-timeline.json` | Fig 03, "Concurrency is not CPU parallelism" |
| A policy or algorithm whose effect shows up as state evolving step by step: LIFO vs FIFO, backoff, eviction, load balancing, circuit breaker states, rollout phases | `stepper` | `assets/example-stepper.json` | Fig 04A/04B, "LIFO sends new work back to the slow process" |
| Resource counts or cost across variants of one flow: connections, threads, pods, shards, before / middle / after of a pooling or multiplexing change | `lanes` | `assets/example-lanes.json` | Fig 05, "The same requests, fewer connections" |

Quick rules:
- If the sentence you would write under the figure starts "By decoupling / routing / splitting / adding ...", use `flow`.
- If it is about *waiting*, use `timeline`: lanes show who waits on whom, and a High-vs-Low panel pair makes a cause visible.
- If it is "X behaves differently when Y happens", use `stepper`: three tabs (setup, mechanism, outcome).
- If it is "the same traffic, now with N instead of M", use `lanes` with one panel per variant, ordered before, then each improvement.
- A bug fix often earns a `stepper` (or `timeline`) for the broken behavior and a second `stepper` for the fixed behavior, named `04A` and `04B` the way the source does. A refactor or migration earns one `flow`.
- Trivial changes (typo, rename, dependency bump) get no figure.

## 2. Commands

```bash
python scripts/render_diagram.py spec.json --out-dir mr/ --name figure-01 --png
```

Produces `figure-01.svg` and `figure-01.html` (animated and interactive: Pause/Replay work, stepper tabs are clickable and auto-advance) and, with `--png`, `figure-01.png` (static frame, buttons hidden). Attach the PNG to the MR, since MR UIs do not run animation or scripts in images.

Flags: `--scale 2` (PNG pixel density; the default gives a PNG about 1900 px wide), `--freeze-at SECONDS` (which animation frame the PNG captures), `--step N` (stepper: 1-based step shown in the PNG; default is the last step, the outcome), `--keep-buttons`. PNG output needs Python `playwright` with Chromium. If it is missing, ship the SVG and HTML and say so.

Always open the PNG after rendering and fix clipping, overlaps and crowded rows before delivering. For a stepper, render with `--step 1` and `--step 3` and look at both.

## 3. Shared fields

```json
{ "kind": "flow|timeline|stepper|lanes",
  "figure": "01",
  "topic": "Connection pooling",
  "title": "Pooled connections spread load evenly",
  "subtitle": "By returning connections in FIFO order, we keep ...",
  "legend": [{"shape": "square|circle|ring|dash", "color": "req|res|cpu|net|wait|#hex", "label": "Request"}],
  "buttons": ["replay", "pause"] }
```

`figure` may carry a suffix (`"04A"`). `topic` is upper-cased into the label (`FIGURE 01 · CONNECTION POOLING`). Default widths: flow 1940, timeline 1790, stepper 1620, lanes 1176 (the source figures' native sizes). Colors by name: `req` blue `#6f9cf7`, `res` mint `#9be8c6`, `cpu` periwinkle `#8fa9ee`, `net` deep blue `#5b7bd6`, `wait` pale blue `#bcd0f5`. Subtitles wrap automatically; keep them under about 200 characters.

Legend: `flow` gets Request and Response by default. The other kinds get none unless you supply one. For `timeline`, the legend names the segment kinds. Use at most three entries.

## 4. `flow`

Nodes and edges on an absolute canvas. The header ends at y≈333; keep content between y 360 and height minus 40.

```json
{ "kind": "flow", "buttons": ["pause"], "width": 1940, "height": 920,
  "groups": [{"id": "svc", "x": 555, "y": 438, "w": 582, "h": 387}],
  "labels": [{"text": "Client", "x": 243, "y": 565}],
  "nodes":  [{"id": "a", "label": "api-gateway", "sub": "process 1", "x": 64, "y": 592, "w": 358, "h": 83}],
  "edges":  [{"from": "a", "to": "b", "fromSide": "r", "toSide": "l", "viaX": 808,
              "step": 1, "arrows": "both", "particles": true}] }
```

- `groups`: outlined box around the system the MR owns or changes. `labels`: mono caps captions centered at `x`, 25-35 px above what they name.
- `nodes`: `label` (sans 20), optional `sub` (mono bold under the label, such as `process 2` or `HTTP/1`). Typical size 186-358 x 83-84. Use equal widths within a column.
- `edges`: sides are `r l t b` (default `r` to `l`). When the ends differ in y, the route is orthogonal with a vertical segment at `viaX` (or `viaY` for top/bottom). Fan-outs share one `viaX` so they draw a single bus. `via: [[x,y],...]` gives custom routing. `arrows`: `both` (default), `end`, `start`, `none`. `step`: hop order; requests animate 1 to N and responses back N to 1. `particles: false` for config or control-plane links.
- Layout: left to right. Callers on the left (x≈64), the system we own in the middle inside a group box, external resources on the right (flush to x≈1885, caps label above, no box). Rows align on shared baselines, 102 px apart. At most 3 replicas and 9-12 nodes in total. Edges never cross nodes.

## 5. `timeline`

Panels of lanes over a shared time axis, with a moving scrubber that reveals segments as it passes. Use two panels to contrast a good case and a bad case ("Low CPU work", "High CPU work"). One panel is fine for a single scenario.

```json
{ "kind": "timeline",
  "legend": [{"shape":"circle","color":"cpu","label":"CPU request/response processing"},
             {"shape":"circle","color":"net","label":"Python network read/write"},
             {"shape":"circle","color":"wait","label":"Wait for Cosmos"}],
  "axis": {"max": 40, "ticks": [0,10,20,30,40], "label": "Time →"},
  "panels": [{
    "title": "High CPU work", "note": "Long Python steps keep ready responses waiting",
    "status": "Python: request C · CPU request processing",
    "thread": {"label": "Python thread", "segments": [{"t":0,"d":6,"kind":"cpu","text":"A"}]},
    "lanes": [{"label": "Request A", "sub": "Ready · blocked", "marker": 12.4,
               "segments": [{"t":0,"d":6,"kind":"cpu","text":"CPU request"}]}],
    "footer": "A: response already at the socket, but Python cannot receive it yet." }] }
```

- Time is in abstract "units": `t` is the start and `d` the duration. Say "illustrative units" in prose. Never invent real milliseconds.
- Segment `kind`: `cpu` (periwinkle: work on the CPU or the single worker), `net` (deep blue: I/O read/write), `wait` (pale blue, dashed: waiting on something external), `blocked` (dashed outline: ready but cannot run), `plain` (black cell with white text: future or unstarted work). `text` shows in mono caps when it fits.
- `thread`: the shared-resource row (the thread, the lock, the connection). `lanes`: one per request or task. A gap before the first segment draws a dotted lead-in automatically.
- `marker`: the time at which something becomes ready (a response arrives). Draws a white diamond and tick on the lane.
- `status` is one sentence about the frozen frame. `footer` is the panel's takeaway, with a number if there is one ("All requests finished at 11.6 units.").
- The contrast between two panels is the point. Keep the same lanes and the same axis in both. At most 2 panels and 4 lanes per panel.

## 6. `stepper`

Three tabs walk a policy through initial state, mechanism, and outcome inside a single card.

```json
{ "kind": "stepper",
  "card": {"title": "Client process", "note": "B3 = connection 3 to server process B. Each process has three connections."},
  "steps": [{
    "tab": "Initial burst", "caption": "An initial burst reaches A, B, and the slower process C.",
    "pool": {"first": "Returned first", "last": "Returned last", "slots": 9,
             "filled": [{"text":"B3"},{"text":"A3"},{"text":"B2"},{"text":"A2","hl":true}],
             "status": "Connections return A2"},
    "servers": [{"name":"A","count":0,"reqs":0,"arrows":{"down":true,"up":true,"ring":true}},
                {"name":"C","count":3,"reqs":3,"banner":"Overloaded","arrows":{"down":true,"up":true,"hl":"down"}}],
    "queue": {"total": 18, "filled": 9} }] }
```

- `pool`: the ordered queue of pills at the top (any ordered resource: connections, workers, tokens). `filled` pills fill left to right and `hl` outlines one in blue. `first`/`last` rename the end labels. `status` is the centered mono line below.
- `servers`: two to four columns. `count` is the big number, `reqs` how many request slots are filled (from the bottom), `slots` the slot count (default 3), `banner` a blue status bar ("Overloaded"). `arrows`: `down`/`up` booleans, `hl: "down"|"up"` colors one blue, `ring: true` puts a response ring on the up arrow.
- `queue`: the dashed strip of pending work; `filled` of `total` boxes are blue.
- Naming: the "bad" policy is `04A`, the fix is `04B`. Titles state the finding ("LIFO sends new work back to the slow process", "FIFO breaks the connection-reuse feedback loop"). The 04B subtitle names the cost honestly ("FIFO maintains more active connections after a burst, but balances workloads fairly across all servers.").
- Always exactly three steps, tab names like `Initial burst`, `Reusing connections`, `Outcome`, each with a one-sentence `caption`.

## 7. `lanes`

Stacked panels, each a miniature of the same flow, with a footer count line. Use it to show cost or capacity dropping across variants.

```json
{ "kind": "lanes", "response_style": "ring",
  "legend": [{"shape":"square","color":"req","label":"Request"},{"shape":"ring","label":"Response"},{"shape":"dash","label":"Idle keep-alive"}],
  "panels": [{
    "title": "Envoy pools HTTP/1 connections",
    "desc": "One shared pool grows to the fleet's peak: 6 connections.",
    "mode": "pooled",
    "headers": [{"text":"Python pods","x":203},{"text":"Shared pool","x":543},{"text":"6 open connections","x":788},{"text":"Storage","x":1047}],
    "rows": [{"label":"Pod 1","sub":"1 active · peak 3","active":1,"peak":3,"incoming":false}],
    "mid": {"label":"Envoy","sub":"HTTP/1 → HTTP/1"}, "open": 6,
    "storage": "Azure Cosmos DB",
    "footer": "6 busy · 0 idle · 6 concurrent upstream requests" }] }
```

- `mode`: `direct` (each row owns `peak` lines straight to storage; the first `active` are solid with particles, the rest dashed idle keep-alive), `pooled` (rows feed a `mid` box, which opens `open` lines to storage), `multiplexed` (the `mid` box holds one line carrying `streams` interleaved particles).
- Rows are the callers (pods, workers, clients), at most 6. `incoming: true` draws a solid lead-in with a request particle. `headers` are mono caps column labels with x centers; copy the x values from the template.
- Every panel carries the same rows so the eye compares counts. The footer states busy / idle / concurrent counts in the same format in every panel. The numbers must be derivable from the diff or the config; otherwise say "illustrative" in the subtitle.
- Order panels before, then each step of the improvement. Two or three panels.

## 8. Content rules (all kinds)

1. Real identifiers only: names of services, hosts, config keys, queues, functions, taken from the diff or repo. Do not invent components.
2. Draw only what the MR is about. Omit auth, metrics and logging unless they are the change.
3. Numbers on a figure follow the same rule as the prose: derived from the diff, config, tests, or the user, or the figure says "illustrative".
4. No decorative icons, no extra colors, no gradients. Dark theme only.
5. One idea per figure. If you need a second idea, make a second figure.
6. Caps mono is for labels, status and axes; sans is for titles, node names and sentences.

## 9. Captions

- **Label** (automatic): `FIGURE NN · TOPIC`. Number figures in order from 01, and use `04A`/`04B` for a before/after pair.
- **Title**: a specific, declarative statement of the finding, sentence case, no period. Source examples: "Simplified Habitat request flow", "Concurrency is not CPU parallelism", "LIFO sends new work back to the slow process", "The same requests, fewer connections".
- **Subtitle**: one or two sentences. For design changes, begin "By ..." and name the benefit ("By decoupling the storage logic into a standalone service, we established a single point of control..."). For explanations, state the mechanism and its consequence in plain declarative sentences.
- In MR Markdown, put the figure after the paragraph that introduces it, with alt text equal to the label plus title and an italic caption line beneath that restates the title and subtitle in one or two sentences.

## 10. Checklist

1. Kind chosen from the table, not from taste.
2. Title states a finding; subtitle states a mechanism or a "By ..." benefit.
3. All labels are real identifiers; any figure numbers are traceable or marked illustrative.
4. Rendered PNG opened and inspected (every step of a stepper). No clipped text, no edge through a node.
5. Delivered: PNG for the MR, plus SVG and HTML originals.

## Animation behaviour (renderer v2)

- Use the `.html` output when animation matters. It includes an interactive playback range for seeking. Inline `.svg` also retains the SVG Replay/Pause controls and animation.
- Flow and lane diagrams animate request/response particles on the paths. Timeline segments reveal in time and the cursor traverses the axis. The playback range seeks the SVG animation clock.
- Stepper diagrams auto-advance every 3.2 seconds, permit manual tab selection, render moving request and response particles on server arrows, and support replay, pausing, and selecting a step via the HTML range.
- Motion respects `prefers-reduced-motion` by starting paused. For a static reviewer-facing screenshot, use `--png`, optionally `--step` or `--freeze-at`.
- The figures are visual reproductions inspired by the article, not extracted source components. Exact animation curves, timings, trajectories, and source artwork have not been established to be pixel-for-pixel identical to OpenAI's site. Don't describe them as verified 1:1 parity.

# Prose style: the Habitat voice

This is the voice of the "Rapidly scaling online storage" engineering post, adapted for merge request descriptions. Match it closely. A reader who has read that post should feel the same author wrote this MR.

The voice is a senior engineer explaining a hard-won decision to peers. It is confident without hype, specific without padding, and honest about what the change costs. It is never a changelog.

Example sentences below are written for MR contexts so you can see the moves transplanted. Do not copy them. Reuse the shape, not the words.

## Contents
1. Voice in one paragraph
2. The moves (with before/after)
3. Section headers
4. Vocabulary
5. Rhythm and mechanics
6. Banned patterns
7. Length
8. Self-check

---

## 1. Voice in one paragraph

First person plural ("we"), active verbs, plain words. Past tense for what happened (the incident, the investigation, the decision). Present tense for how the system behaves now. Every claim that can carry a number carries one. Trade-offs are named out loud, with what got worse and why that is acceptable *at this particular moment*. Occasional idiom ("squeeze as much juice out of", "buy time", "blast radius") gives it a human pulse, about one per section, never stacked.

## 2. The moves

### 2.1 Open with the consequence, not the mechanism
Start from what the user, the product, or the on-call engineer experiences. Then narrow to the system. Use the parallel-conditional pair to set stakes when it is true:

> If those requests are slow, the product feels slow. If those requests fail, the product stops working entirely.

MR version:
- Weak: "Adds retry logic to the payments client."
- Strong: "Every checkout depends on the payments client returning within a second. When it hangs, the cart hangs. When it fails, the customer sees an error instead of a receipt."

### 2.2 Say what it is, then say what this change does to it
After the stakes, one short paragraph defines the component in a sentence a new teammate could follow, then states what this MR changes and the headline result. Introduce unfamiliar terms once, with a parenthetical gloss: "Statsig (a tool that manages feature flags, and can be used to run A/B tests)", "(known as a 'thundering herd')".

### 2.3 Put real numbers on everything that can hold one
Sizes, rates, percentages, counts, latencies, before/after ratios. Use "more than", "over", "almost", "up to" for honest rounding. Compare with ratios ("6x more CPU efficient and 15x more memory efficient").

- Pull numbers from the diff (files, lines, config values, timeouts, pool sizes, thresholds), from commit messages, or from the user.
- **Never invent a measurement.** If the diff does not contain a benchmark and the user did not give one, state the change qualitatively and say what was or was not measured ("We haven't load-tested this yet; the numbers above come from the unit-level benchmark in `bench_pool.py`"). A missing number is acceptable. A fabricated one is not.
- Config values count as numbers: "from a 60-second poll with no jitter to a 5-minute poll with ±20% jitter".

### 2.4 Tell the root cause as a short detective story
When the MR fixes a bug or a performance problem, narrate it in this order, past tense, one beat per sentence or two:

1. What we saw (symptom, with a number if possible)
2. What we expected instead
3. What we suspected
4. How we tested the suspicion
5. What we found
6. The fix, and how small or large it turned out to be

Anchor phrases from the source voice: "We discovered this in a chance incident where...", "We suspected X was to blame and tested this suspicion by...", "which indeed limited the degradation and confirmed our investigation direction", "The fix was straightforward once profiling helped us root cause the issue."

If the diff gives no evidence of the investigation, do not fabricate one. Describe the mechanism of the bug and the fix instead, and ask the user for the discovery story if it would add value.

### 2.5 Name the trade-off, then own it
Every non-trivial design choice gets one sentence stating what it costs, one stating why it is still right, and, when relevant, one stating when to revisit. Source phrasings: "an explicit tradeoff", "we think is the right tradeoff to make at this particular moment", "a strategic incursion of technical debt", "a calculated wager".

- Weak: "This improves performance."
- Strong: "This adds one network hop to every read, which costs about 2 ms at p50. We think that is the right trade for a single place to enforce access control, and we'll revisit it if the hop shows up in tail latency."

### 2.6 Put the claim in the first sentence, then support it
Each paragraph is one idea, two to five sentences. The first sentence states the point. The rest give the mechanism, the number, or the consequence. Follow a long, explanatory sentence with a short one that lands the conclusion:

> That bet eventually proved correct.

### 2.7 Contrast what it replaces with what it does
Use "Instead of X, we...", "Rather than allowing X, Y exposes Z", "By doing X, we get Y". A "By ..." sentence is also the house style for figure subtitles.

### 2.8 Be explicit about limits and next steps
Say what the change deliberately does not do and why. Close with a forward pointer when one exists: "In a follow-up change, we'll move the remaining callers and delete the old path." Pointers are specific, never vague ("more to come").

### 2.9 Prefer prose to bullets
The source is almost entirely paragraphs. Use bullets only for genuinely discrete enumerations (rollout steps, test commands, a list of affected services). Never bullet the narrative.

### 2.10 Stat strip
When there are two or three real, measured, headline numbers, render them as a stat strip directly under the opening, the way the source shows "70M+ requests per second / 1B+ people each week / 500 PB+ data". In Markdown:

```markdown
| **6x** | **15x** | **95%** |
|:--|:--|:--|
| CPU efficiency | memory efficiency | of production traffic migrated |
```

Skip the strip when there are fewer than two real numbers. Never pad it.

## 3. Section headers

Sentence case. Specific. A noun phrase or a gerund that says what the section is *about*, not what *kind* of section it is. Source examples: "Tracking the asyncio delay", "Balancing loads and managing connection pools", "Avoiding flooding downstream resources", "Why Habitat does less", "Launching a Python service at scale", "Migrate from Python to Rust".

Patterns that work:
- Gerund: "Reducing tail latency in feature flag refreshes", "Avoiding duplicate webhook deliveries"
- "Why X does Y": "Why the cache sits behind the router"
- Imperative-as-title for the core move: "Move retries into the client"
- Plain noun phrase naming the thing: "Tracking the asyncio delay"

Never use: Overview, Background, Context, Summary, Changes, Details, Implementation, Misc, Notes. Those are labels for kinds of content. The voice names the content.

Use two to five headers for a typical MR. The opening has no header.

## 4. Vocabulary

Use these where they are true. Do not force them. About one idiom per section.

Systems framing: blast radius, single point of control, single chokepoint, fan-out, fan-in, hot path, tail latencies (p99 and higher), latency cliffs, feedback loop, metastable failure, thundering herd, escape hatch, predictable constant-work requests, mission-critical traffic, centrally enforce, steady state, in-flight requests, backward-compatible, decoupling X into Y.

Decision framing: tactical decisions, sequencing, buy time, foundational investments, fending off capacity crunches, calculated wager, strategic incursion of technical debt, at this particular moment, unblocking product developers, prioritize the more immediate challenges.

Colloquial seasoning: squeeze as much juice out of, chance incident, well-acquainted with, throw caution to the wind, swing for, it was finally time to.

Connectives: "As a result,", "In practice,", "Combined, this meant", "Once a ... became ..., ...", "One side effect of ...", "Before X, we ...", "Instead of ...".

## 5. Rhythm and mechanics

- Paragraphs of two to five sentences. Mix lengths. A one-sentence paragraph is allowed for a hinge ("That bet eventually proved correct.").
- Em dashes are fine, used sparingly, for a single aside per paragraph at most.
- Parentheticals carry definitions and counts. Do not nest them.
- Code identifiers, flags, env vars, file paths, and metric names go in backticks.
- Numbers: digits for anything measured (6x, 20 million), words for small counts in running prose ("two engineers", "three regions"). Use "x" for multipliers (6x). Spell out units on first use, abbreviate after.
- Past tense for events and decisions, present tense for current behavior, future ("we'll") only for follow-ups.
- Refer to "this change" or "we", never "this MR" / "this PR" in the prose. "In this change, we..." is fine once.
- Figure captions: label (`FIGURE 01 · TOPIC`), headline sentence, then one "By ..." takeaway sentence. See `diagram-style.md`.

## 6. Banned patterns

- Emojis, exclamation marks, rhetorical questions.
- Hype and filler: seamless, game-changing, blazing, cutting-edge, "simply", "just", "basically", "various", "a number of", "several" without a count.
- Hedging that sounds unsure about your own change: "I think", "hopefully", "might help".
- Changelog voice: "Added X. Fixed Y. Updated Z." Turn each into a reason plus an effect.
- "This PR/MR adds...", "In this PR, I...", first-person singular.
- Bare "improves"/"enhances"/"optimizes" with no number or mechanism.
- Section headers that name a *kind* of content (see section 3).
- Bullet lists as the main narrative.
- Fabricated numbers, incidents, benchmarks, or quotes. Ever.

## 7. Length

Scale with the change, not with the diff's line count.

| Change | Target |
|:--|:--|
| Small fix, one file | 120-220 words, no stat strip, one header at most, optional small figure |
| Typical feature or refactor | 350-600 words, 2-3 headers, one figure |
| Architectural change | 600-900 words, 3-5 headers, one or two figures, stat strip if numbers exist |

Cut sentences that restate the diff. Reviewers can read the diff. They cannot read your reasons.

## 8. Self-check before finishing

1. Does the first sentence name a consequence someone feels, not a file or function?
2. Is every number traceable to the diff, a commit, or the user? Remove any that are not.
3. Is there one explicit trade-off sentence, with a reason it is acceptable now?
4. Do headers say what the section is about, in sentence case?
5. Is the narrative in paragraphs, in "we" voice, with no changelog list?
6. Is there at most one idiom per section, and none stacked?
7. Does the figure caption follow label, headline, "By ..." takeaway?
8. Would a peer finishing the source post believe the same author wrote this?

# MR description + figure

Turn the current branch into a merge request description a peer would want to read, plus one or two figures in the same house style (request flow, timeline, tabbed walkthrough, or stacked comparison). The voice and the diagram style are fixed on purpose: they are modeled exactly on one reference engineering post, so every MR the user opens reads like the same author wrote it.

Two reference files carry the detail. Read both before writing:
- `references/prose-style.md`: the voice, with the moves, header patterns, vocabulary, banned patterns, length targets, and a self-check.
- `references/diagram-style.md`: how to choose among the four figure kinds, the JSON spec for each, content and layout rules, and captions.

## Workflow

### 1. Diff against the base

```bash
bash scripts/collect_diff.sh            # auto-detects origin/HEAD, main, master, develop
bash scripts/collect_diff.sh main --max-lines 2500
bash scripts/collect_diff.sh --working-tree   # include uncommitted work
```

The script prints the branch, merge-base, commits, changed files, diffstat, and the diff (lockfiles and build output excluded, capped by `--max-lines`). It uses the merge-base, so the diff shows only what this branch introduced.

If the output is truncated or a hunk is unclear, read the touched files directly (`git show HEAD:path`, or open the file). Do not describe code you have not read. If the user names a different base, pass it.

### 2. Find the story before writing

Answer these four in a scratch note. They become the skeleton.

1. **Consequence:** who or what feels the problem today? (user, product, on-call, cost)
2. **Mechanism:** what does the system do, and what does this change do to it?
3. **Evidence:** which numbers exist? Pull from the diff (timeouts, pool sizes, thresholds, counts), commit messages, test or benchmark output, and the user. List what you do *not* have.
4. **Trade-off:** what does this cost, and why is that acceptable now?

If the commit log tells an investigation story (symptom, suspicion, test, finding, fix), use it. If the evidence for a number or an incident is missing, either ask the user one short question or leave the claim qualitative. Never invent numbers, incidents, or benchmarks.

### 3. Write the description

Follow `references/prose-style.md` strictly. Skeleton:

```markdown
# <MR title: sentence case, specific, at most 72 characters, no type prefix>

<Opening: consequence first, then what the system is, then what this change does and its headline result. 2-4 sentences, no header.>

<Optional stat strip, only with two or more real numbers.>

## <Specific header naming what the section is about>
<Paragraphs in "we" voice. Mechanism, story, numbers.>

![FIGURE 01 · TOPIC: Title](figure-01.png)
*Figure 01 · <Title>. <By ... takeaway sentence.>*

## <Header for the trade-off or the "why X does less" section>
<Name the cost, why it is right at this moment, when to revisit.>

## <Header for testing and rollout, if there is something to say>
<How it was verified, how it ships, what to watch. Short bullets are allowed here for commands or steps.>

<Closing forward pointer if one exists: what follow-up change comes next.>
```

Scale to the change: a one-file fix gets roughly 150 words and at most one header; an architectural change gets 600-900 words. See the length table in the style guide.

### 4. Build the figures

Follow `references/diagram-style.md`. There are four figure kinds, each with a template in `assets/`. Pick by what the change does, using the chooser table in the style guide:

| Change | Kind | Template |
|:--|:--|:--|
| New or rewired components, proxy, split, migration | `flow` | `assets/example-spec.json` |
| Ordering, blocking, concurrency, retries, queueing delay | `timeline` | `assets/example-timeline.json` |
| A policy whose effect unfolds over steps (LIFO vs FIFO, backoff, eviction) | `stepper` | `assets/example-stepper.json` |
| Resource counts across before/after variants of one flow | `lanes` | `assets/example-lanes.json` |

Default to one figure, allow two (they may be different kinds), and skip the figure entirely for trivial changes (typo, dependency bump, formatting, comment-only), saying so in one line.

1. Choose the kind(s). Copy the matching template to `figure-01.json` and edit it with real identifiers from the repo and the diff. Numbers on a figure follow the same evidence rule as the prose, or the subtitle says "illustrative".
2. Render each:
   ```bash
   python scripts/render_diagram.py figure-01.json --out-dir mr/ --name figure-01 --png
   ```
3. Open every PNG with the image viewer (for a stepper, render and inspect `--step 1` and `--step 3`) and fix clipped text, overlapping edges, and crowded groups. Re-render until clean.

### 5. Deliver

Write `description.md` into the same `mr/` folder as the figure files. Then:
- In claude.ai or Cowork: copy `mr/` to `/mnt/user-data/outputs/` and call `present_files` with `description.md` first, then each figure's `.png`, `.svg`, `.html`.
- In a local repo: leave `mr/` in the working tree root (add it to `.git/info/exclude` so it is not committed) and print the paths.

In the final message, keep it short: where the files are, the one place where a number was left qualitative because the evidence was missing (if any), and a reminder that the PNG must be attached to the MR by drag-and-drop or upload, after which the markdown image path should be replaced with the uploaded URL. MR UIs cannot read local relative paths, and CLI tools (`glab`, `gh`) do not upload images.

## Rules that matter most

- The voice is the product. If a sentence could appear in a generic changelog, rewrite it with a consequence, a number, or a trade-off.
- Numbers come from evidence. No evidence, no number.
- One figure by default, two at most. Real names only. Nine to twelve nodes at most.
- Reviewers read the diff; the description gives them the reasons. Do not walk the diff file by file.
- If the user gives a style correction ("shorter", "less dramatic", "more detail on the rollback"), apply it within the voice. Do not drop the voice.

### Animated output

Prefer linking the generated `.html` for interactive diagrams: it has a seek bar as well as SVG Replay/Pause controls. In a stepper, tabs auto-advance, clicking tabs selects a stage, and server paths carry request/response markers. Use a PNG only for an inline MR image; PNG cannot reproduce animation. See `references/diagram-style.md` for playback behaviour and the parity limitation.
